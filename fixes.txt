Simplified logic:
  1. Parse the target and replacement.
  2. Remove all duplicate definitions of the same target within the lexical chain.
  3. Insert a single new definition if new_code is nonempty.
"""
lexical_chain = lexical_chain or []

# deletion requested
if not new_code.strip():
    if not target_name:
        raise ValueError("Cannot delete without target_name.")
    stripped, _ = remove_block(content, target_name, lexical_chain)
    return stripped, True

# infer target info
replacement_module = cst.parse_module(new_code)
node, inferred_kind, inferred_name = _extract_replacement_node(replacement_module, target_name)
if node is None:
    raise ValueError("new_code must contain at least one function, class, or assignment.")
kind = kind or inferred_kind
target_name = target_name or inferred_name

# parse original content
module = cst.parse_module(content)
body: List[cst.BaseStatement] = list(module.body)

# helper to match lexical chain
def in_chain(stack: list[str]) -> bool:
    return stack == lexical_chain

# walk and filter body recursively to drop duplicates
def filter_nodes(nodes: list[cst.BaseStatement], stack: list[str]) -> list[cst.BaseStatement]:
    out: list[cst.BaseStatement] = []
    for n in nodes:
        if isinstance(n, cst.ClassDef):
            new_stack = stack + [n.name.value]
            if in_chain(new_stack) and n.name.value == target_name and kind == "class":
                continue
            n = n.with_changes(body=filter_nodes(n.body.body, new_stack))
            out.append(n)
        elif isinstance(n, cst.FunctionDef):
            if in_chain(stack) and n.name.value == target_name and kind in (None, "def", "func"):
                continue
            out.append(n)
        elif isinstance(n, cst.SimpleStatementLine):
            name = _extract_assignment_name(n)
            if in_chain(stack) and name == target_name and kind == "assign":
                continue
            out.append(n)
        else:
            out.append(n)
    return out

filtered = filter_nodes(body, [])

# append replacement
replacement_code = cst.Module(body=[node]).code
new_module = cst.parse_module(cst.Module(body=filtered).code + "\n" + replacement_code)
return new_module.code, True
